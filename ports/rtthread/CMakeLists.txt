if(NOT TARGET cagent_core)
    message(FATAL_ERROR "Add the cAgent source directory before the RT-Thread port")
endif()
option(AGENT_RTTHREAD_RUNTIME "Build RT-Thread 5.1+ Runtime" OFF)
option(AGENT_RTTHREAD_TRANSPORT "Build RT-Thread WebClient Transport" OFF)
option(AGENT_RTTHREAD_FILE_STORE "Build RT-Thread DFS/POSIX file store" OFF)
option(AGENT_RTTHREAD_NO_SYMLINKS "Trusted DFS mount has no symbolic links" OFF)
set(AGENT_RTTHREAD_SDK_TARGET "" CACHE STRING "Application SDK target carrying RT-Thread/WebClient includes and libraries")
if(AGENT_RTTHREAD_SDK_TARGET AND NOT TARGET ${AGENT_RTTHREAD_SDK_TARGET})
    message(FATAL_ERROR "AGENT_RTTHREAD_SDK_TARGET must name an existing application target")
endif()

foreach(part runtime transport)
    string(TOUPPER "${part}" upper)
    if(AGENT_RTTHREAD_${upper})
        add_library(cagent_rtthread_${part} STATIC ${part}/src/${part}.c)
        add_library(cagent::rtthread_${part} ALIAS cagent_rtthread_${part})
        target_include_directories(cagent_rtthread_${part} PUBLIC ${part}/include
            PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../src")
        target_link_libraries(cagent_rtthread_${part} PUBLIC cagent_core)
        if(AGENT_RTTHREAD_SDK_TARGET)
            target_link_libraries(cagent_rtthread_${part} PUBLIC ${AGENT_RTTHREAD_SDK_TARGET})
        endif()
        set_target_properties(cagent_rtthread_${part} PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
    endif()
endforeach()

if(AGENT_RTTHREAD_FILE_STORE)
    if(NOT TARGET cagent_posix_file_store)
        message(FATAL_ERROR "RT-Thread file store requires AGENT_BUILD_FILE_STORE and AGENT_BUILD_POSIX_FILE_STORE")
    endif()
    if(AGENT_RTTHREAD_NO_SYMLINKS)
        target_compile_definitions(cagent_posix_file_store PRIVATE AGENT_POSIX_FILE_STORE_NO_SYMLINKS=1)
    endif()
    add_library(cagent_rtthread_file_store STATIC storage/src/file_store.c)
    add_library(cagent::rtthread_file_store ALIAS cagent_rtthread_file_store)
    target_include_directories(cagent_rtthread_file_store PUBLIC storage/include)
    target_link_libraries(cagent_rtthread_file_store PUBLIC cagent_posix_file_store)
    if(AGENT_RTTHREAD_SDK_TARGET)
        target_link_libraries(cagent_posix_file_store PUBLIC ${AGENT_RTTHREAD_SDK_TARGET})
    endif()
    set_target_properties(cagent_rtthread_file_store PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
endif()
