cmake_minimum_required(VERSION 3.16)
project(rtthread_port_contract C)
get_filename_component(root "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
add_library(test_sdk INTERFACE)
target_include_directories(test_sdk INTERFACE "${root}/tests/ports/rtthread/fake")
set(AGENT_RTTHREAD_SDK_TARGET test_sdk CACHE STRING "" FORCE)
add_subdirectory("${root}" core)
add_subdirectory("${root}/ports/rtthread" rtthread)
if(AGENT_RTTHREAD_RUNTIME)
    add_executable(runtime_contract ../runtime_contract.c)
    target_link_libraries(runtime_contract PRIVATE cagent::rtthread_runtime)
endif()
if(AGENT_RTTHREAD_TRANSPORT)
    add_executable(transport_contract ../transport_contract.c)
    target_include_directories(transport_contract PRIVATE "${root}/src")
    target_link_libraries(transport_contract PRIVATE cagent::rtthread_transport)
endif()
if(AGENT_RTTHREAD_FILE_STORE)
    add_executable(file_contract "${root}/tests/ports/file_store_binding.c")
    target_compile_definitions(file_contract PRIVATE TEST_RTTHREAD)
    target_link_libraries(file_contract PRIVATE cagent::rtthread_file_store)
endif()
