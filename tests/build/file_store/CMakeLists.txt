cmake_minimum_required(VERSION 3.16)
project(cagent_file_build_contract C)
get_filename_component(AGENT_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(FILE_STORE_PLATFORM "posix" CACHE STRING "Build contract platform")
option(FILE_STORE_WITH_JSONL "Include JSONL and its shared-file bridge" OFF)
option(FILE_STORE_REUSE "Reuse prebuilt common targets in NuttX assembly" OFF)

if(FILE_STORE_PLATFORM STREQUAL "idf")
    # Exercise component source/usage assembly without claiming a native SDK build.
    function(idf_component_register)
        cmake_parse_arguments(COMP "" "" "SRCS;INCLUDE_DIRS;PRIV_INCLUDE_DIRS;REQUIRES;PRIV_REQUIRES" ${ARGN})
        add_library(${_component_name} STATIC ${COMP_SRCS})
        target_include_directories(${_component_name} PUBLIC ${COMP_INCLUDE_DIRS}
            PRIVATE ${COMP_PRIV_INCLUDE_DIRS})
        target_link_libraries(${_component_name} PUBLIC ${COMP_REQUIRES})
        set(COMPONENT_LIB ${_component_name} PARENT_SCOPE)
    endfunction()
    set(CONFIG_AGENT_FILE_STORE ON)
    set(CONFIG_AGENT_POSIX_FILE_STORE ON)
    set(CONFIG_AGENT_SESSION_JSONL ${FILE_STORE_WITH_JSONL})
    set(CONFIG_AGENT_PORT_ESPIDF_FILE_STORE ON)
    set(_component_name cagent)
    add_subdirectory("${AGENT_ROOT}" core)
    set(_component_name cagent_espidf_port)
    add_subdirectory("${AGENT_ROOT}/ports/espidf" port)
    add_executable(file_store_smoke "${AGENT_ROOT}/tests/ports/file_store_binding.c")
    target_compile_definitions(file_store_smoke PRIVATE TEST_ESPIDF=1)
    target_link_libraries(file_store_smoke PRIVATE cagent_espidf_port)
    if(FILE_STORE_WITH_JSONL)
        target_compile_definitions(file_store_smoke PRIVATE TEST_JSONL=1)
    endif()
elseif(FILE_STORE_PLATFORM STREQUAL "openvela")
    function(nuttx_add_library name kind)
        add_library(${name} ${kind} ${ARGN})
    endfunction()
    set(CONFIG_AGENT_FILE_STORE ON)
    set(CONFIG_AGENT_PORT_OPENVELA_FILE_STORE ON)
    if(FILE_STORE_REUSE)
        set(AGENT_BUILD_FILE_STORE ON CACHE BOOL "" FORCE)
        set(AGENT_BUILD_POSIX_FILE_STORE ON CACHE BOOL "" FORCE)
        set(AGENT_BUILD_JSON_CODEC ${FILE_STORE_WITH_JSONL} CACHE BOOL "" FORCE)
        set(AGENT_BUILD_SESSION_JSONL ${FILE_STORE_WITH_JSONL} CACHE BOOL "" FORCE)
        add_subdirectory("${AGENT_ROOT}" core)
    endif()
    add_subdirectory("${AGENT_ROOT}/ports/openvela" port)
    add_executable(file_store_smoke "${AGENT_ROOT}/tests/ports/file_store_binding.c")
    target_compile_definitions(file_store_smoke PRIVATE TEST_OPENVELA=1)
    target_link_libraries(file_store_smoke PRIVATE cagent_openvela_file_store)
    if(FILE_STORE_WITH_JSONL)
        target_compile_definitions(file_store_smoke PRIVATE TEST_JSONL=1)
        target_link_libraries(file_store_smoke PRIVATE cagent_session_jsonl_files)
    endif()
else()
    set(AGENT_BUILD_FILE_STORE ON CACHE BOOL "" FORCE)
    set(AGENT_BUILD_POSIX_FILE_STORE ON CACHE BOOL "" FORCE)
    set(AGENT_BUILD_SESSION_JSONL ${FILE_STORE_WITH_JSONL} CACHE BOOL "" FORCE)
    set(AGENT_BUILD_JSON_CODEC ${FILE_STORE_WITH_JSONL} CACHE BOOL "" FORCE)
    add_subdirectory("${AGENT_ROOT}" core)
    add_executable(file_store_smoke smoke.c)
    target_link_libraries(file_store_smoke PRIVATE cagent::posix_file_store)
    if(FILE_STORE_WITH_JSONL)
        target_compile_definitions(file_store_smoke PRIVATE TEST_JSONL=1)
        target_link_libraries(file_store_smoke PRIVATE cagent::session_jsonl_files)
    endif()
endif()
