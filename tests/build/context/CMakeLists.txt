cmake_minimum_required(VERSION 3.16)
project(cagent_context_build_contract C)
get_filename_component(AGENT_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
option(CONTEXT_BUILD_IDF "Exercise IDF assembly with a mock component registrar" OFF)
if(CONTEXT_BUILD_IDF)
    function(idf_component_register)
        cmake_parse_arguments(COMP "" "" "SRCS;INCLUDE_DIRS;PRIV_INCLUDE_DIRS;PRIV_REQUIRES" ${ARGN})
        add_library(cagent_core STATIC ${COMP_SRCS})
        target_include_directories(cagent_core PUBLIC ${COMP_INCLUDE_DIRS} PRIVATE ${COMP_PRIV_INCLUDE_DIRS})
        set(COMPONENT_LIB cagent_core PARENT_SCOPE)
    endfunction()
    set(CONFIG_AGENT_SESSION_RAM ON)
else()
    set(AGENT_BUILD_SESSION_RAM ON CACHE BOOL "" FORCE)
endif()
add_subdirectory("${AGENT_ROOT}" core)
add_executable(context_contract "${AGENT_ROOT}/tests/context/contract.c")
target_include_directories(context_contract PRIVATE "${AGENT_ROOT}/src")
target_link_libraries(context_contract PRIVATE cagent_core)
if(NOT CONTEXT_BUILD_IDF)
    target_link_libraries(context_contract PRIVATE cagent::session_ram)
endif()
